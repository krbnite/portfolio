#      the youtube_daily_video_metrics table (e.g., tot num of likes).

#      where: <daily schedule> = <min> <hr> <monthday:*> <month:*> <weekday:*> 
#------------------------------------------------------------------------------

50 7 * * * /home/ubuntu/Automations/youtube/daily/video_metrics/youtube_daily_video_metrics.py channel_02 --not-a-test && /home/ubuntu/Automations/youtube/daily/top10/youtube_daily_top10_and_channel_metrics.py channel_02 --not-a-test
0  8 * * * /home/ubuntu/Automations/youtube/daily/video_metrics/youtube_daily_video_metrics.py channel_03 --not-a-test && /home/ubuntu/Automations/youtube/daily/top10/youtube_daily_top10_and_channel_metrics.py channel_03 --not-a-test
